cmake_minimum_required(VERSION 3.24)
project(dng_linear_frame VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(DNG_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/dng-sdk-1.7.1" CACHE PATH "Unpacked Adobe DNG SDK 1.7.1")
if(NOT APPLE)
  message(FATAL_ERROR "This adapter currently supports macOS (Adobe Converter) only")
endif()
if(NOT EXISTS "${DNG_SDK_ROOT}/dng_sdk/source/dng_flags.h")
  message(FATAL_ERROR "Unpack the licensed Adobe SDK first; see README.md")
endif()
# Prefer matching Unix dependency headers/libraries over unrelated macOS
# frameworks (e.g. Mono bundles old libpng/libjpeg headers).
set(CMAKE_FIND_FRAMEWORK LAST)
find_package(PNG REQUIRED)
find_package(JPEG REQUIRED)
find_package(ZLIB REQUIRED)
find_path(JXL_INCLUDE_DIR jxl/decode.h REQUIRED)
find_library(JXL_LIBRARY jxl REQUIRED)
find_library(JXL_THREADS_LIBRARY jxl_threads REQUIRED)
file(GLOB sdk_sources CONFIGURE_DEPENDS "${DNG_SDK_ROOT}/dng_sdk/source/*.cpp")
list(FILTER sdk_sources EXCLUDE REGEX "/(dng_validate)\\.cpp$")
# Source selection mirrors XMPCore/build/CMakeListsCommon.txt, old DOM API.
# Keep all generated objects out of the read-only SDK vendor tree.
set(xmp "${DNG_SDK_ROOT}/xmp/toolkit")
file(GLOB xmp_sources CONFIGURE_DEPENDS "${xmp}/XMPCore/source/XMP*.cpp" "${xmp}/XMPCore/source/WXMP*.cpp" "${xmp}/third-party/zuid/sources/*.cpp")
add_library(xmp_core STATIC ${xmp_sources}
 "${xmp}/XMPCore/source/CoreObjectFactoryImpl.cpp" "${xmp}/XMPCore/source/ExpatAdapter.cpp" "${xmp}/XMPCore/source/ParseRDF.cpp"
 "${xmp}/source/UnicodeConversions.cpp" "${xmp}/source/XML_Node.cpp" "${xmp}/source/XMP_LibUtils.cpp"
 "${xmp}/XMPCore/third-party/expat/public/lib/xmlparse.c" "${xmp}/XMPCore/third-party/expat/public/lib/xmlrole.c" "${xmp}/XMPCore/third-party/expat/public/lib/xmltok.c")
target_include_directories(xmp_core PRIVATE "${xmp}" "${xmp}/public/include" "${xmp}/XMPCore/third-party/expat/public/lib" "${xmp}/XMPCore/resource/mac")
target_compile_definitions(xmp_core PRIVATE MAC_ENV=1 XMP_64=1 XMP_StaticBuild=1 ENABLE_CPP_DOM_MODEL=0 XML_STATIC=1 HAVE_EXPAT_CONFIG_H=1 XMP_COMPONENT_INT_NAMESPACE=AdobeXMPCore_Int BUILDING_XMPCORE_LIB=1 BUILDING_XMPCORE_AS_STATIC=1)
add_library(adobe_dng STATIC ${sdk_sources})
target_include_directories(adobe_dng PUBLIC "${DNG_SDK_ROOT}/dng_sdk/source" ${JXL_INCLUDE_DIR} PRIVATE "${xmp}/public/include")
target_compile_definitions(adobe_dng PUBLIC qMacOS=1 qDNGUseStdInt=1 qDNGUseXMP=1 qDNGXMPFiles=0 qDNGXMPDocOps=0 XMP_StaticBuild=1 XMP_64=1 MAC_ENV=1 ENABLE_CPP_DOM_MODEL=0 qDNGUseLibJPEG=1 qDNGSupportVC5=0 qDNGValidateTarget=0 qDNGPrintMessages=0)
target_link_libraries(adobe_dng PUBLIC xmp_core JPEG::JPEG ZLIB::ZLIB ${JXL_LIBRARY} ${JXL_THREADS_LIBRARY} "-framework CoreFoundation" "-framework CoreServices")
add_library(frame_core src/tiff.cpp src/color.cpp src/decode.cpp)
target_include_directories(frame_core PUBLIC src)
target_link_libraries(frame_core PUBLIC adobe_dng)
target_compile_options(frame_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(dng-linear-frame src/main.cpp src/process.cpp src/png.cpp)
target_link_libraries(dng-linear-frame PRIVATE frame_core PNG::PNG)
target_compile_options(dng-linear-frame PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(frame-unit-tests tests/unit.cpp)
  target_link_libraries(frame-unit-tests PRIVATE frame_core)
  add_test(NAME color-quantization COMMAND frame-unit-tests)
  add_executable(sdk-probe tests/sdk_probe.cpp)
  target_link_libraries(sdk-probe PRIVATE frame_core)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME routing-sdk-semantics COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:dng-linear-frame> $<TARGET_FILE:sdk-probe> ${CMAKE_CURRENT_BINARY_DIR}/test-artifacts)
endif()
